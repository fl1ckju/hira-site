function sahne(gizle, goster, odak, sonra, yazi) {
  perde(yazi, () => {
    gizle.forEach(e => { e.hidden = true; });
    goster.forEach(e => { e.hidden = false; });
    scrollTo({ top: 0, behavior: 'instant' });
    if (sonra) sonra();
    if (odak) odak.focus({ preventScroll: true });
  });
}
